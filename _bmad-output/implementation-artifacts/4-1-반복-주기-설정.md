# Story 4.1: 반복 주기 설정

Status: review

## Story

사용자(Catty)로서,
할일을 생성하거나 수정할 때 반복 주기(매일/매주/매월)를 설정하는 것을 원한다,
루틴 할일을 한 번만 등록하면 자동으로 반복되도록.

## Acceptance Criteria

1. **반복 주기 옵션 제공**: 할일 생성 또는 수정 폼이 열려 있을 때, "반복 없음 / 매일 / 매주 (요일 선택) / 매월 (날짜 선택)" 선택지가 제공된다.

2. **반복 주기 DB 저장**: 반복 주기가 설정된 할일을 저장할 때, `todos` 테이블의 `recurrence_type` / `recurrence_days` / `recurrence_day_of_month` 컬럼에 반복 주기 정보가 저장된다.

3. **매주 요일 정보 저장**: 매주 반복을 선택할 때 요일을 지정하면, 지정된 요일 정보가 `recurrence_days` 컬럼에 저장된다 (0=월, 1=화, ..., 6=일, 쉼표 구분 문자열).

4. **매월 날짜 정보 저장**: 매월 반복을 선택할 때 날짜를 지정하면, 지정된 날짜 정보가 `recurrence_day_of_month` 컬럼에 저장된다 (1-31 정수).

5. **기존 할일 동작 무영향**: 반복 설정이 없는 기존 할일은 새 컬럼이 NULL이며, 대시보드·검색·완료 처리 등 기존 기능이 정확히 동일하게 동작한다.

```python
# backend/services/todos.py
from typing import Any, Optional

def create_todo(
    user_id: str,
    supabase,
    title: str,
    category_id: Optional[str] = None,
    priority: Optional[str] = None,
    due_date: Optional[str] = None,
    recurrence_type: Optional[str] = None,
    recurrence_days: Optional[str] = None,
    recurrence_day_of_month: Optional[int] = None,
) -> dict:
    data: dict[str, Any] = {
        "user_id": user_id,
        "title": title,
        "is_completed": False,
    }
    if category_id is not None:
        data["category_id"] = category_id
    if priority is not None:
        data["priority"] = priority
    if due_date is not None:
        data["due_date"] = due_date
    if recurrence_type is not None:
        data["recurrence_type"] = recurrence_type
    if recurrence_days is not None:
        data["recurrence_days"] = recurrence_days
    if recurrence_day_of_month is not None:
        data["recurrence_day_of_month"] = recurrence_day_of_month

    response = (
        supabase.table("todos")
        .insert(data)
        .select("*, categories(name)")
        .single()
        .execute()
    )
    return response.data


def update_todo_content(
    todo_id: str,
    user_id: str,
    supabase,
    title: Optional[str] = None,
    category_id: Optional[str] = None,
    priority: Optional[str] = None,
    due_date: Optional[str] = None,
    recurrence_type: Optional[str] = None,
    recurrence_days: Optional[str] = None,
    recurrence_day_of_month: Optional[int] = None,
) -> dict:
    updates: dict[str, Any] = {}
    if title is not None:
        updates["title"] = title
    if category_id is not None:
        updates["category_id"] = category_id
    if priority is not None:
        updates["priority"] = priority
    if due_date is not None:
        updates["due_date"] = due_date
    if recurrence_type is not None:
        updates["recurrence_type"] = recurrence_type
        # recurrence 해제 시 null로 초기화
    if recurrence_days is not None:
        updates["recurrence_days"] = recurrence_days
    if recurrence_day_of_month is not None:
        updates["recurrence_day_of_month"] = recurrence_day_of_month

    if not updates:
        raise ValueError("업데이트할 필드가 없습니다")

    response = (
        supabase.table("todos")
        .update(updates)
        .eq("id", todo_id)
        .eq("user_id", user_id)
        .select("*, categories(name)")
        .single()
        .execute()
    )
    return response.data
```

**주의**: `recurrence_type`을 'none'으로 받는 대신 프론트에서 "반복 없음" 선택 시 필드를 null로 전송하거나 아예 포함하지 않아야 함. recurrence_type=None → DB에 NULL 저장 = 반복 없음.

    recurrence_day_of_month: Optional[int] = None

class TodoCreateResponse(BaseModel):

    priority: Optional[str] = None

        recurrence_type=body.recurrence_type,
        recurrence_days=body.recurrence_days,
        recurrence_day_of_month=body.recurrence_day_of_month,
    )

@router.put("/{todo_id}", response_model=TodoCreateResponse)
def update_todo_content_endpoint(
    todo_id: str,
    body: TodoContentUpdateRequest,
    current_user=Depends(get_current_user),
    supabase=Depends(get_supabase),
):
    user_id = getattr(current_user, "id", None) or getattr(current_user, "sub", None)
    if not user_id:
        raise HTTPException(status_code=401)
    row = update_todo_content(
        todo_id=todo_id,
        user_id=user_id,
        supabase=supabase,
        title=body.title,
        category_id=body.category_id,
        priority=body.priority,
        due_date=body.due_date,
        recurrence_type=body.recurrence_type,
        recurrence_days=body.recurrence_days,
        recurrence_day_of_month=body.recurrence_day_of_month,
    )
    if not row:
        raise HTTPException(status_code=404, detail="할일을 찾을 수 없습니다")
    return row
